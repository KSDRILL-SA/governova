§ Prisma Schema — Locked v2.0
prisma/schema.prisma — Auth Domain
model User {
  id              String    @id @default(cuid())
  email           String    @unique
  passwordHash    String    @map("password_hash")
  name            String?
  role            Role      @default(USER)
  emailVerifiedAt DateTime? @map("email_verified_at")
  createdAt       DateTime  @default(now()) @map("created_at")
  updatedAt       DateTime  @updatedAt @map("updated_at")
  deletedAt       DateTime? @map("deleted_at")

  sessions        Session[]
  auditLogs       AuditLog[]
  failedAttempts  FailedLoginAttempt[]
  accountLocks    AccountLock[]
  securityAlerts  SecurityAlert[]

  @@index([email])
  @@index([deletedAt])
  @@map("users")
}

model Session {
  id            String   @id @default(cuid())
  userId        String   @map("user_id")
  tokenHash     String   @unique @map("token_hash")
  expiresAt     DateTime @map("expires_at")
  lastUsedAt    DateTime @default(now()) @map("last_used_at")
  requestCount  Int      @default(0) @map("request_count")
  lastUpdatedAt DateTime @default(now()) @map("last_updated_at")
  createdAt     DateTime @default(now()) @map("created_at")
  ip            String?
  userAgent     String?  @map("user_agent")
  deviceId      String?  @map("device_id")

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([expiresAt])
  @@index([lastUsedAt])
  @@map("sessions")
}

model AuditLog {
  id         String   @id @default(cuid())
  userId     String?  @map("user_id")
  action     String
  metadata   Json?
  createdAt  DateTime @default(now()) @map("created_at")
  dateKey    String   @map("date_key")
  durationMs Int?     @map("duration_ms")
  ip         String?
  userAgent  String?  @map("user_agent")

  user User? @relation(fields: [userId], references: [id])

  @@index([userId, createdAt])
  @@index([dateKey])
  @@map("audit_logs")
}

model FailedLoginAttempt {
  id        String   @id @default(cuid())
  email     String
  ip        String
  attempts  Int      @default(1)
  firstAt   DateTime @default(now()) @map("first_at")
  lastAt    DateTime @updatedAt @map("last_at")
  alertedAt DateTime? @map("alerted_at")

  @@unique([email, ip])
  @@index([email, lastAt])
  @@index([ip, lastAt])
  @@map("failed_login_attempts")
}

model AccountLock {
  email       String   @id
  lockedUntil DateTime @map("locked_until")
  reason      String
  createdAt   DateTime @default(now()) @map("created_at")
  metadata    Json?

  user User? @relation(fields: [email], references: [email])

  @@index([lockedUntil])
  @@map("account_locks")
}

model SecurityAlert {
  id        String   @id @default(cuid())
  userId    String   @map("user_id")
  type      String
  severity  String
  metadata  Json?
  sentAt    DateTime @default(now()) @map("sent_at")
  readAt    DateTime? @map("read_at")

  user User @relation(fields: [userId], references: [id])

  @@index([userId, sentAt])
  @@map("security_alerts")
}

model RateLimit {
  id        String   @id @default(cuid())
  ip        String
  endpoint  String
  createdAt DateTime @default(now()) @map("created_at")

  @@index([ip, endpoint, createdAt])
  @@map("rate_limits")
}

enum Role {
  USER
  ADMIN
  CREATOR
}
§ JWT & Cookie Configuration — Locked v2.0
JWT Payload Structure
interface JWTPayload {
  // Standard claims
  sub: string;        // user_id
  email: string;      // user email
  role: string;       // "USER" | "ADMIN" | "CREATOR"
  
  // Timestamps
  iat: number;        // issued at
  exp: number;        // absolute expiry (7 days)
  
  // Session tracking
  jti: string;        // JWT ID (matches session.id)
  
  // Security metadata (v2 addition)
  ip: string;         // Client IP at login
  ua: string;         // User agent hash
  ver: number;        // Token version (for rotation)
}
■
Cookie Configuration: HttpOnly=true, Secure=true (HTTPS only), SameSite=Lax, Path="/", Max-Age=604800 (7 days)
⚡
Optimized Session Lifecycle: 95% reduction in database writes. Update lastUsedAt every 5 minutes OR after 100 requests — otherwise async increment requestCount only.
